// ETS LedgerPro — Google Sheets Sync (v4) // Paste ALL of this into Extensions → Apps Script, then Deploy → New deployment → Web App // Execute as: Me | Who has access: Anyone const SECRET = 'YOUR_TOKEN_HERE'; // ← replace with any password you choose const SID = 'YOUR_SHEET_ID_HERE'; // ← replace with your Google Sheet ID // ─── ENTRY POINTS ─────────────────────────────────────────── function doGet(e) { const p = e.parameter; if (p.token !== SECRET) return out({ error: 'Bad token' }); if (p.action === 'test') return out({ ok: true, msg: 'ETS LedgerPro connected', t: new Date().toISOString() }); if (p.action === 'load') { try { const ss = SpreadsheetApp.openById(SID); const blob = getDataSheet(ss).getRange('A1').getValue(); return out({ ok: true, data: blob ? JSON.parse(blob) : {} }); } catch(e) { return out({ error: e.message }); } } return out({ error: 'unknown action' }); } function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.token !== SECRET) return out({ error: 'Bad token' }); const ss = SpreadsheetApp.openById(SID); const db = body.db; const ds = getDataSheet(ss); ds.getRange('A1').setValue(JSON.stringify(db)); ds.getRange('A2').setValue('Saved: ' + new Date().toLocaleString()); writeTab(ss, 'Clients', db.clients, ['id','name','contact','email','phone','ein','defaultFee','notes']); writeTab(ss, 'Entries', db.entries, ['id','number','clientId','clientName','port','mode','date','hts','description','value','dutyRate','duty','mpf','hmf','brokerFee','status']); writeTab(ss, 'Invoices', db.invoices, ['id','number','clientId','clientName','issueDate','dueDate','amount','status','notes']); writeTab(ss, 'AP', db.apInvoices||[], ['id','vendor','invoiceNo','category','entryNumber','date','dueDate','amount','billable','status','paidDate','memo']); writeTab(ss, 'Bookkeeping', db.bkgTransactions, ['id','date','inv','type','cat','net','tax','fees','total','desc','note','auto','entryRef']); writeTab(ss, 'PipelineQueue', db.plQueue, ['_id','_email_date','_classification','entry_number','awb_number','shipper_name','consignee_name','hts_code','dutiable_value_usd','port_of_entry','eta_date','_queueStatus']); writeTab(ss, 'PipelineResults', db.plResults, ['_id','_email_date','_classification','entry_number','awb_number','shipper_name','consignee_name','hts_code','dutiable_value_usd','port_of_entry','eta_date','_queueStatus']); return out({ saved: true, entries: (db.entries||[]).length, t: new Date().toISOString() }); } catch(err) { return out({ error: err.message }); } } function getDataSheet(ss) { return ss.getSheetByName('ETS_Data') || ss.insertSheet('ETS_Data'); } function writeTab(ss, name, rows, cols) { if (!rows || !rows.length) return; let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name); sheet.clearContents(); const data = [cols].concat(rows.map(r => cols.map(c => { const v = r[c]; if (v == null) return ''; if (typeof v === 'object') return JSON.stringify(v); return v; }))); sheet.getRange(1, 1, data.length, cols.length).setValues(data); sheet.getRange(1, 1, 1, cols.length).setBackground('#1A3A6B').setFontColor('#FFFFFF').setFontWeight('bold'); } function out(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); } function testWrite() { const ss = SpreadsheetApp.openById(SID); let t = ss.getSheetByName('_Test'); if (!t) t = ss.insertSheet('_Test'); t.getRange('A1').setValue('✓ Script is working!'); t.getRange('A2').setValue('Ran at: ' + new Date().toString()); Logger.log('testWrite complete — check _Test tab in your sheet'); }
